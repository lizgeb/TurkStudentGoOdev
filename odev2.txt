SELECT 
    invoice_id, 
    total AS old_total, 
    total * 2 AS new_total 
FROM Invoice 
ORDER BY new_total DESC;

SELECT 
    invoice_id, 
    LEFT(billing_address, 3)||RIGHT(billing_address, 4) AS "new_address"
FROM Invoice
WHERE DATE_PART('year', invoice_date) = 2013 
AND DATE_PART('month', invoice_date) = 8;
